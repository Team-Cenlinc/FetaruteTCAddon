  /** 一次 attempt 的完整产物（反射进 prepare/attempt，绕过搜索，严格失败也拿得到冲突列表）。 */
  private record Probe(
      int headway,
      Timetable timetable,
      TimetableConflictChecker.Report conflicts,
      List<ResourceRepair.Yield> yields,
      List<TimetableBuildResult.TripShift> shifts,
      List<TimetableBuildResult.DroppedTrip> dropped,
      List<TerminalSerializer.TerminalReport> terminals,
      List<String> phaseNotes,
      TurnaroundTable turnarounds,
      List<ConflictAbsorption.Residual> residuals,
      String failure,
      long millis) {}

  @SuppressWarnings("unchecked")
  private static Probe attemptOf(Member member, int headway, int maxWait) throws Exception {
    TimetableBuildOptions options = optionsOf(headway, true, 4, maxWait);
    TimetableBuilder builder = new TimetableBuilder();
    java.lang.reflect.Method resolve =
        TimetableBuilder.class.getDeclaredMethod(
            "resolveTurnarounds", TimetableBuilder.BuildInput.class, TimetableBuildOptions.class);
    resolve.setAccessible(true);
    TurnaroundTable turnarounds = (TurnaroundTable) resolve.invoke(null, member.input(), options);
    TimetableBuildOptions resolved = options.withTurnaround(turnarounds);
    java.lang.reflect.Method prepare =
        TimetableBuilder.class.getDeclaredMethod(
            "prepare", TimetableBuilder.BuildInput.class, TimetableBuildOptions.class, List.class);
    prepare.setAccessible(true);
    long t0 = System.nanoTime();
    Object prepared;
    try {
      prepared = prepare.invoke(builder, member.input(), resolved, new ArrayList<>());
    } catch (java.lang.reflect.InvocationTargetException ex) {
      return new Probe(headway, null, TimetableConflictChecker.Report.none(), List.of(), List.of(), List.of(), List.of(), List.of(), turnarounds, List.of(), "prepare: " + ex.getCause().getMessage(), 0);
    }
    java.lang.reflect.Method attempt =
        TimetableBuilder.class.getDeclaredMethod(
            "attempt", prepared.getClass(), TimetableBuildOptions.class, TimetableBuilder.BuildInput.class, Instant.class);
    attempt.setAccessible(true);
    Object a;
    try {
      a = attempt.invoke(builder, prepared, resolved, member.input(), Instant.parse("2026-09-20T00:00:00Z"));
    } catch (java.lang.reflect.InvocationTargetException ex) {
      return new Probe(headway, null, TimetableConflictChecker.Report.none(), List.of(), List.of(), List.of(), List.of(), List.of(), turnarounds, List.of(), "attempt: " + ex.getCause().getMessage(), (System.nanoTime() - t0) / 1_000_000L);
    }
    long millis = (System.nanoTime() - t0) / 1_000_000L;
    return new Probe(
        headway,
        (Timetable) field(a, "timetable"),
        (TimetableConflictChecker.Report) field(a, "conflicts"),
        (List<ResourceRepair.Yield>) field(a, "yields"),
        (List<TimetableBuildResult.TripShift>) field(a, "shifts"),
        (List<TimetableBuildResult.DroppedTrip>) field(a, "dropped"),
        (List<TerminalSerializer.TerminalReport>) field(a, "terminals"),
        (List<String>) field(a, "phaseNotes"),
        turnarounds,
        (List<ConflictAbsorption.Residual>) field(a, "residuals"),
        null,
        millis);
  }

  private static Object field(Object record, String accessor) throws Exception {
    java.lang.reflect.Method m = record.getClass().getDeclaredMethod(accessor);
    m.setAccessible(true);
    return m.invoke(record);
  }

  private static String clock(int relative) {
    return TimetableCsvExporter.clock(5 * 3600 + relative);
  }

  private static void summaryWithResiduals(String label, Probe p) {
    summary(label, p);
    if (p.failure() != null) {
      return;
    }
    List<ConflictAbsorption.Residual> ok = ConflictAbsorption.absorbable(p.residuals());
    List<ConflictAbsorption.Residual> no = ConflictAbsorption.unabsorbable(p.residuals());
    log(
        String.format(
            "        残余 可吸收 %-4d 不可吸收 %-4d  %s  最长可吸收等待 %ds",
            ok.size(),
            no.size(),
            ConflictAbsorption.countByVerdict(no),
            ok.stream().mapToInt(ConflictAbsorption.Residual::waitSeconds).max().orElse(0)));
  }

  private static void summary(String label, Probe p) {
    if (p.failure() != null) {
      log(String.format("  %s h=%-5d 失败 (%d ms): %s", label, p.headway(), p.millis(), p.failure()));
      return;
    }
    Map<String, Integer> kinds = new LinkedHashMap<>();
    for (TimetableConflictChecker.Conflict c : p.conflicts().conflicts()) {
      kinds.merge(c.kind().name(), 1, Integer::sum);
    }
    log(String.format(
        "  %s h=%-5d 班次 %-4d 交路 %-3d 取消 %-3d 冲突 %-5d %s 让车 %-4d (最长 +%ds) 偏离 %-4d (%d ms) 折返 %s",
        label, p.headway(), p.timetable().trips().size(), p.timetable().duties().size(), p.dropped().size(),
        p.conflicts().conflicts().size(), kinds, p.yields().size(), ResourceRepair.maxWait(p.yields()), p.shifts().size(), p.millis(), p.turnarounds().describe()));
    if (!p.conflicts().clean()) {
      log("        top " + p.conflicts().topResources(6));
    }
  }

  private static String routeOf(String code) {
    if (code.matches("D\\d+")) {
      return "待命";
    }
    if (code.matches("D\\d+-CREATE")) {
      return "出库走行";
    }
    if (code.matches("D\\d+-RETURN")) {
      return "回库走行";
    }
    int dash = code.lastIndexOf('-');
    return dash > 0 ? code.substring(0, dash) : code;
  }

  private static void pairs(Probe p, int limit) {
    Map<String, Integer> pairs = new java.util.TreeMap<>();
    for (TimetableConflictChecker.Conflict c : p.conflicts().conflicts()) {
      pairs.merge(c.kind().name() + " " + routeOf(c.first()) + " × " + routeOf(c.second()), 1, Integer::sum);
    }
    log("        route 对（前 " + limit + "）：");
    pairs.entrySet().stream().sorted(Map.Entry.<String, Integer>comparingByValue().reversed()).limit(limit)
        .forEach(en -> log("          " + en.getValue() + "  " + en.getKey()));
  }

  private static void table(Probe p, int maxDuties) {
    Timetable t = p.timetable();
    Map<UUID, TimetableTrip> byId = new HashMap<>();
    t.trips().forEach(x -> byId.put(x.id(), x));
    log("        相位: " + p.phaseNotes());
    for (TimetableRoutePlan plan : t.routePlans()) {
      log(String.format("        plan %-14s %-9s %s→%s 全程 %ds 停靠 %d 折返 %ds", plan.routeCode(), plan.kind(), shortNode(plan.originNodeId()), shortNode(plan.terminalNodeId()), plan.totalRunSeconds(), plan.stops().size(), p.turnarounds().secondsFor(plan.routeId())));
    }
    int shown = 0;
    for (VehicleDuty d : t.duties()) {
      if (shown++ >= maxDuties) {
        log("        … 另有 " + (t.duties().size() - maxDuties) + " 条交路");
        break;
      }
      StringBuilder sb = new StringBuilder();
      sb.append(String.format("        %s 出库 %s(%s) ", d.dutyCode(), clock(d.plannedStartSecondOfDay() - 18000), d.createRouteId().flatMap(t::routePlan).map(TimetableRoutePlan::routeCode).orElse("-")));
      for (UUID id : d.tripIds()) {
        TimetableTrip trip = byId.get(id);
        if (trip == null) continue;
        int run = t.routePlan(trip.routeId()).map(TimetableRoutePlan::totalRunSeconds).orElse(0);
        sb.append(String.format("| %s %s→%s ", trip.tripCode(), clock(trip.departureSecondOfDay() - 18000), clock(trip.departureSecondOfDay() - 18000 + run)));
      }
      sb.append(String.format("| 回库 %s(%s) %s", clock(d.returnSecondOfDay() - 18000), d.returnRouteId().flatMap(t::routePlan).map(TimetableRoutePlan::routeCode).orElse("-"), d.closeReason()));
      log(sb.toString());
    }
    int c = 0;
    for (TimetableConflictChecker.Conflict conflict : p.conflicts().conflicts()) {
      if (c++ >= 10) break;
      log("          · " + conflict.describe(s -> clock(s)));
    }
  }

  private static void runOne(Member member, int headway, boolean strict, int maxTrips) {
    TimetableBuildOptions options = optionsOf(headway, strict, maxTrips);
    long started = System.nanoTime();
    TimetableBuildResult result =
        new TimetableBuilder().build(member.input(), options, Instant.parse("2026-09-20T00:00:00Z"));
    report(member.lineCode(), headway, result, System.nanoTime() - started);
  }

  private static String shortNode(String nodeId) {
    return nodeId == null ? "?" : nodeId.replace("SURC:", "");
  }

  private static void report(
      String label, int headway, TimetableBuildResult result, long nanos) {
    long ms = nanos / 1_000_000L;
    if (result.timetable().isEmpty()) {
      log(
          String.format(
              "  h=%-5d 失败 (%d ms) %s",
              headway, ms, String.join(" | ", result.warnings())));
      return;
    }
    Timetable table = result.timetable().orElseThrow();
    StringBuilder shares = new StringBuilder();
    for (WeightedTripAllocator.ShareReport share : result.shares()) {
      shares
          .append(share.key())
          .append(String.format("=%.0f/%.0f%% ", share.targetShare() * 100, share.achievedShare() * 100));
    }
    log(
        String.format(
            "  h=%-5d → 实际 %-5d 班次 %-4d 交路 %-3d 峰值车 %-3d 取消 %-4d 冲突 %-4d 让车 %-3d (%d ms)",
            headway,
            result.effectiveHeadwaySeconds(),
            table.trips().size(),
            result.dutyCount(),
            result.peakConcurrentVehicles(),
            result.droppedTrips().size(),
            result.conflictsAtTarget().size(),
            result.yields().size(),
            ms));
    log("        份额 " + shares.toString().trim());
    for (TerminalSerializer.TerminalReport t : result.terminals()) {
      log(
          String.format(
              "        端点 %s 经过 %d 次 占用 %.0f%% 无处等待 %d 截断 %d",
              t.group(), t.visits(), t.utilization() * 100, t.nowhereToWait(), t.truncated()));
    }
    Map<String, Integer> byReason = new LinkedHashMap<>();
    for (TimetableBuildResult.DroppedTrip d : result.droppedTrips()) {
      byReason.merge(d.reason().name(), 1, Integer::sum);
    }
    if (!byReason.isEmpty()) {
      log("        取消原因 " + byReason);
    }
  }

  private record Member(
      TimetableBuilder.BuildInput input, String lineCode, Set<UUID> ownRouteIds) {}

  private static Member memberOf(
      StorageProvider provider,
      RouteDefinitionCache cache,
      Company company,
      Operator operator,
      Line line,
      List<Route> operatorLegs,
      RailGraph graph,
      RailTravelTimeModel model,
      boolean pureLegsOnly) {
    Map<UUID, Route> routes = new LinkedHashMap<>();
    Set<UUID> own = new java.util.LinkedHashSet<>();
    for (Route r : provider.routes().listByLine(line.id())) {
      routes.put(r.id(), r);
      own.add(r.id());
    }
    for (Route r : operatorLegs) {
      if (pureLegsOnly && !own.contains(r.id()) && hasIntermediateStop(provider, r)) {
        // 别的线的「带客」CREATE/RETURN：Q1 会把它当班次排进我的表。这一档把它排除，量污染有多大。
        continue;
      }
      routes.putIfAbsent(r.id(), r);
    }
    List<TimetableBuilder.RouteInput> inputs = new ArrayList<>();
    for (Route route : routes.values()) {
      RouteDefinition def = cache.findById(route.id()).orElse(null);
      if (def == null) {
        log("  SKIP no definition: " + route.code());
        continue;
      }
      List<RouteStop> stops =
          provider.routeStops().listByRoute(route.id()).stream()
              .sorted(Comparator.comparingInt(RouteStop::sequence))
              .toList();
      inputs.add(
          new TimetableBuilder.RouteInput(
              route.id(),
              route.code(),
              route.operationType(),
              readWeight(route),
              def,
              stops,
              Optional.empty(),
              Optional.empty(),
              false,
              readGroup(route)));
    }
    UUID timetableId =
        UUID.nameUUIDFromBytes(
            ("PROBE-" + line.code()).getBytes(java.nio.charset.StandardCharsets.UTF_8));
    TimetableBuilder.BuildInput input =
        new TimetableBuilder.BuildInput(
            timetableId,
            company.id(),
            operator.id(),
            line.id(),
            "PROBE-" + line.code(),
            "probe",
            inputs,
            graph,
            model,
            Optional.empty());
    return new Member(input, line.code(), own);
  }

  private static TimetableBuildOptions optionsOf(int headway, boolean strict, int maxTrips) {
    return optionsOf(headway, strict, maxTrips, TimetableBuildOptions.Repair.DEFAULT_MAX_WAIT_SECONDS);
  }

  private static TimetableBuildOptions optionsOf(int headway, boolean strict, int maxTrips, int maxWait) {
    return new TimetableBuildOptions(
        5 * 3600,
        24 * 3600,
        Duration.ofSeconds(headway),
        Duration.ofSeconds(TimetableBuildOptions.DEFAULT_DWELL_SECONDS),
        new VehicleDutyPlanner.Limits(
            maxTrips, VehicleDutyPlanner.Limits.DEFAULT_MAX_DURATION_SECONDS,
            TurnaroundTable.none()),
        "",
        ZoneId.of("UTC"),
        Duration.ofSeconds(TimetableBuildOptions.DEFAULT_SEPARATION_SECONDS),
        strict,
        Map.of(),
        new TimetableBuildOptions.Repair(
            Duration.ofSeconds(maxWait), Duration.ofSeconds(Math.max(300, maxWait))));
  }

  /** 中途有 STOP 的 CREATE/RETURN 是带客 route（与 ServiceGroupClassifier 同一判据）。 */
  private static boolean hasIntermediateStop(StorageProvider provider, Route route) {
    List<RouteStop> stops =
        provider.routeStops().listByRoute(route.id()).stream()
            .sorted(Comparator.comparingInt(RouteStop::sequence))
            .toList();
    for (int i = 1; i + 1 < stops.size(); i++) {
      if (stops.get(i).passType() == org.fetarute.fetaruteTCAddon.company.model.RouteStopPassType.STOP) {
        return true;
      }
    }
    return false;
  }

  private static int readWeight(Route route) {
    Object raw = route.metadata() == null ? null : route.metadata().get("spawn_weight");
    if (raw instanceof Number number) {
      return Math.max(0, number.intValue());
    }
    return 1;
  }

  private static Optional<String> readGroup(Route route) {
    Object raw = route.metadata() == null ? null : route.metadata().get("spawn_group");
    return raw instanceof String text && !text.isBlank() ? Optional.of(text) : Optional.empty();
  }
