  private record Summary(
      int total,
      int self,
      int real,
      Map<TimetableConflictChecker.Kind, Integer> realByKind,
      Map<String, Integer> realByResource,
      int realChtPlatform,
      int realChtSingle,
      String chtBetween,
      String chtFinal,
      long chtStaySum,
      String gaps2C,
      String samples,
      String chains) {}

  private static Summary summarize(
      Timetable tt,
      Map<UUID, TimetableConflictChecker.RouteProfile> profiles,
      TimetableConflictChecker.GraphIndex index,
      int zero,
      TimetableConflictChecker.Report report) {
    Map<UUID, TimetableTrip> byId = new HashMap<>();
    Map<String, String> dutyOfCode = new HashMap<>();
    Map<UUID, String> dutyCodeById = new HashMap<>();
    for (VehicleDuty duty : tt.duties()) {
      dutyCodeById.put(duty.id(), duty.dutyCode());
      dutyOfCode.put(duty.dutyCode(), duty.dutyCode());
      dutyOfCode.put(duty.dutyCode() + "-CREATE", duty.dutyCode());
      dutyOfCode.put(duty.dutyCode() + "-RETURN", duty.dutyCode());
    }
    for (TimetableTrip trip : tt.trips()) {
      byId.put(trip.id(), trip);
      trip.dutyId().map(dutyCodeById::get).ifPresent(d -> dutyOfCode.put(trip.tripCode(), d));
    }
    int self = 0;
    int real = 0;
    Map<TimetableConflictChecker.Kind, Integer> realByKind = new TreeMap<>();
    Map<String, Integer> realByResource = new TreeMap<>();
    int chtPlat = 0;
    int chtSingle = 0;
    StringBuilder samples = new StringBuilder();
    int sampled = 0;
    for (TimetableConflictChecker.Conflict c : report.conflicts()) {
      String a = dutyOfCode.get(c.first());
      String b = dutyOfCode.get(c.second());
      if (a != null && a.equals(b)) {
        self++;
        continue;
      }
      real++;
      realByKind.merge(c.kind(), 1, Integer::sum);
      realByResource.merge(c.resource(), 1, Integer::sum);
      if (c.kind() == TimetableConflictChecker.Kind.PLATFORM && c.resource().contains(CHT_GROUP)) {
        chtPlat++;
      }
      if (c.kind() == TimetableConflictChecker.Kind.SINGLE_LINE && c.resource().contains("CHT")) {
        chtSingle++;
      }
      if (sampled < 4) {
        samples.append(c.describe(s -> String.valueOf(s))).append(" ;; ");
        sampled++;
      }
    }
    List<Integer> between = new ArrayList<>();
    List<Integer> fin = new ArrayList<>();
    List<Integer> deps2C = new ArrayList<>();
    StringBuilder chains = new StringBuilder();
    int shown = 0;
    for (VehicleDuty duty : tt.duties()) {
      List<TimetableTrip> chain = duty.tripIds().stream().map(byId::get).toList();
      StringBuilder line = new StringBuilder("      " + duty.dutyCode() + ":");
      for (int i = 0; i < chain.size(); i++) {
        TimetableTrip trip = chain.get(i);
        TimetableRoutePlan plan = tt.routePlan(trip.routeId()).orElseThrow();
        int dep = rel(trip.departureSecondOfDay(), zero);
        int arr = dep + plan.totalRunSeconds();
        if (plan.routeCode().startsWith("WS-2C")) {
          deps2C.add(dep);
        }
        line.append(' ').append(plan.routeCode().substring(3, 5)).append(' ').append(dep).append('>').append(arr);
        if (i + 1 < chain.size()) {
          int wait = rel(chain.get(i + 1).departureSecondOfDay(), zero) - arr;
          if (plan.terminalNodeId().equals(CHT)) {
            between.add(wait);
          }
          line.append(" w").append(wait);
        } else if (duty.returnRouteId().isPresent()) {
          int wait = rel(duty.returnSecondOfDay(), zero) - arr;
          if (plan.terminalNodeId().equals(CHT)) {
            fin.add(wait);
          }
          line.append(" fw").append(wait);
        }
      }
      if (shown < 6) {
        chains.append(line).append('\n');
        shown++;
      }
    }
    TimetableOccupancyProjector.Occupancy occ = TimetableOccupancyProjector.project(tt, profiles, zero);
    long staySum = 0;
    for (TimetableConflictChecker.Stay stay : occ.stays()) {
      if (stay.platform().group().equals(CHT_GROUP)) {
        staySum += stay.to() - stay.from();
      }
    }
    deps2C.sort(null);
    List<Integer> gaps = new ArrayList<>();
    for (int i = 1; i < deps2C.size(); i++) {
      gaps.add(deps2C.get(i) - deps2C.get(i - 1));
    }
    gaps.sort(null);
    String gapText =
        gaps.isEmpty()
            ? "-"
            : "min=" + gaps.get(0) + " med=" + gaps.get(gaps.size() / 2) + " max=" + gaps.get(gaps.size() - 1);
    return new Summary(
        report.conflicts().size(),
        self,
        real,
        realByKind,
        realByResource,
        chtPlat,
        chtSingle,
        stats(between),
        stats(fin),
        staySum,
        gapText,
        samples.toString(),
        chains.toString());
  }

  private record Serialized(
      Timetable timetable, int delayed, long delaySum, int delayMax, int stubOriginDelayed, int stubVisits) {}

  /** 单股道端点按资源串行：到达要等端点空出来（延后整趟车），续班发车 = 到达 + 折返。 */
  private static Serialized serializeAtStubs(
      Timetable tt,
      Map<UUID, TimetableConflictChecker.RouteProfile> profiles,
      TimetableConflictChecker.GraphIndex index,
      int zero,
      int separation) {
    java.util.Set<String> stubs = new java.util.HashSet<>();
    index.platformCapacity().forEach((g, c) -> { if (c == 1) stubs.add(g); });
    Map<UUID, TimetableTrip> byId = new HashMap<>();
    for (TimetableTrip trip : tt.trips()) {
      byId.put(trip.id(), trip);
    }
    List<VehicleDuty> dutyList = tt.duties();
    List<List<TimetableTrip>> chains = new ArrayList<>();
    int[] next = new int[dutyList.size()];
    int[] ready = new int[dutyList.size()];
    int[] returnRel = new int[dutyList.size()];
    for (VehicleDuty duty : dutyList) {
      chains.add(duty.tripIds().stream().map(byId::get).toList());
    }
    Map<UUID, Integer> newDep = new HashMap<>();
    Map<String, Integer> freeAt = new HashMap<>();
    int delayed = 0;
    long delaySum = 0;
    int delayMax = 0;
    int stubOriginDelayed = 0;
    int stubVisits = 0;
    java.util.PriorityQueue<int[]> queue =
        new java.util.PriorityQueue<>(
            (a, b) -> a[0] != b[0] ? Integer.compare(a[0], b[0]) : Integer.compare(a[1], b[1]));
    for (int d = 0; d < dutyList.size(); d++) {
      if (chains.get(d).isEmpty()) continue;
      int nominal = rel(chains.get(d).get(0).departureSecondOfDay(), zero);
      ready[d] = nominal;
      queue.add(new int[] {nominal, d});
    }
    while (!queue.isEmpty()) {
      int[] head = queue.poll();
      int d = head[1];
      List<TimetableTrip> chain = chains.get(d);
      TimetableTrip trip = chain.get(next[d]);
      TimetableRoutePlan plan = tt.routePlan(trip.routeId()).orElseThrow();
      TimetableConflictChecker.RouteProfile profile = profiles.get(trip.routeId());
      int nominal = rel(trip.departureSecondOfDay(), zero);
      boolean originStub = stubs.contains(TimetableConflictChecker.groupOf(plan.originNodeId()));
      int dep = originStub ? ready[d] : Math.max(nominal, ready[d]);
      int arr = dep + plan.totalRunSeconds();
      String terminalGroup = TimetableConflictChecker.groupOf(plan.terminalNodeId());
      boolean terminalStub = stubs.contains(terminalGroup);
      if (terminalStub) {
        int in = approachIn(profile, terminalGroup);
        int earliestArr = freeAt.getOrDefault(terminalGroup, Integer.MIN_VALUE / 2) + in;
        if (arr < earliestArr) {
          int delta = earliestArr - arr;
          dep += delta;
          arr = earliestArr;
          if (originStub) stubOriginDelayed++;
        }
        stubVisits++;
      }
      if (dep != nominal) {
        delayed++;
        delaySum += Math.abs(dep - nominal);
        delayMax = Math.max(delayMax, Math.abs(dep - nominal));
      }
      newDep.put(trip.id(), dep);
      int nextReady = arr + TURNAROUND;
      boolean hasNext = next[d] + 1 < chain.size();
      VehicleDuty duty = dutyList.get(d);
      if (terminalStub) {
        int out;
        if (hasNext) {
          TimetableTrip nt = chain.get(next[d] + 1);
          out = approachOut(profiles.get(nt.routeId()), terminalGroup);
        } else {
          out = duty.returnRouteId().map(profiles::get).map(pr -> approachOut(pr, terminalGroup)).orElse(0);
        }
        freeAt.put(terminalGroup, nextReady + out + separation);
      }
      if (hasNext) {
        next[d]++;
        ready[d] = nextReady;
        TimetableTrip nt = chain.get(next[d]);
        TimetableRoutePlan np = tt.routePlan(nt.routeId()).orElseThrow();
        boolean nextOriginStub = stubs.contains(TimetableConflictChecker.groupOf(np.originNodeId()));
        int key = nextOriginStub ? nextReady : Math.max(rel(nt.departureSecondOfDay(), zero), nextReady);
        queue.add(new int[] {key, d});
      } else {
        returnRel[d] = duty.returnRouteId().isPresent() ? nextReady : arr;
      }
    }
    List<TimetableTrip> trips = new ArrayList<>();
    for (TimetableTrip trip : tt.trips()) {
      Integer dep = newDep.get(trip.id());
      trips.add(
          dep == null
              ? trip
              : new TimetableTrip(
                  trip.id(), trip.timetableId(), trip.routeId(), trip.sequence(), trip.tripCode(),
                  Math.floorMod(zero + dep, 86400), trip.dutyId()));
    }
    List<VehicleDuty> duties = new ArrayList<>();
    for (int d = 0; d < dutyList.size(); d++) {
      VehicleDuty duty = dutyList.get(d);
      int returnRun =
          duty.returnRouteId().flatMap(tt::routePlan).map(TimetableRoutePlan::totalRunSeconds).orElse(0);
      duties.add(
          new VehicleDuty(
              duty.id(), duty.timetableId(), duty.sequence(), duty.dutyCode(), duty.startDepotNodeId(),
              duty.endDepotNodeId(), duty.createRouteId(), duty.returnRouteId(), duty.tripIds(),
              duty.plannedStartSecondOfDay(), Math.floorMod(zero + returnRel[d], 86400),
              Math.floorMod(zero + returnRel[d] + returnRun, 86400), duty.closeReason()));
    }
    Timetable out =
        new Timetable(
            tt.id(), tt.companyId(), tt.operatorId(), tt.lineId(), tt.code(), tt.name(), tt.status(),
            tt.zoneId(), tt.serviceStartSecondOfDay(), tt.serviceEndSecondOfDay(), tt.routePlans(),
            trips, duties, tt.notes(), tt.createdAt(), tt.updatedAt());
    return new Serialized(out, delayed, delaySum, delayMax, stubOriginDelayed, stubVisits);
  }

  /** 进站走行：从最后一个不属于端点站台组的停靠点出发，到终点到达。 */
  private static int approachIn(TimetableConflictChecker.RouteProfile profile, String group) {
    List<TimetableStop> stops = profile.stops();
    int terminalArrival = stops.get(stops.size() - 1).arrivalOffsetSeconds();
    for (int i = stops.size() - 2; i >= 0; i--) {
      if (!TimetableConflictChecker.groupOf(stops.get(i).nodeId().orElse("")).equals(group)) {
        return terminalArrival - stops.get(i).departureOffsetSeconds();
      }
    }
    return terminalArrival;
  }

  /** 出站走行：从起点出发到第一个不属于端点站台组的停靠点到达。 */
  private static int approachOut(TimetableConflictChecker.RouteProfile profile, String group) {
    List<TimetableStop> stops = profile.stops();
    for (int i = 1; i < stops.size(); i++) {
      if (!TimetableConflictChecker.groupOf(stops.get(i).nodeId().orElse("")).equals(group)) {
        return stops.get(i).arrivalOffsetSeconds();
      }
    }
    return 0;
  }

  /** 方案 2 的模拟：从容量 1 站台出发的续班，发车 = 上一班到达 + 折返，不等网格。 */
  private static Timetable anchorAtCapacityOne(
      Timetable tt, TimetableConflictChecker.GraphIndex index, int zero) {
    Map<UUID, TimetableTrip> byId = new HashMap<>();
    for (TimetableTrip trip : tt.trips()) {
      byId.put(trip.id(), trip);
    }
    Map<UUID, TimetableTrip> updated = new HashMap<>(byId);
    List<VehicleDuty> duties = new ArrayList<>();
    for (VehicleDuty duty : tt.duties()) {
      List<TimetableTrip> chain = duty.tripIds().stream().map(byId::get).toList();
      int prevArrival = Integer.MIN_VALUE;
      for (int i = 0; i < chain.size(); i++) {
        TimetableTrip trip = chain.get(i);
        TimetableRoutePlan plan = tt.routePlan(trip.routeId()).orElseThrow();
        int dep = rel(trip.departureSecondOfDay(), zero);
        if (i > 0) {
          int ready = prevArrival + TURNAROUND;
          Integer capacity = index.platformCapacity().get(TimetableConflictChecker.groupOf(plan.originNodeId()));
          if (capacity != null && capacity == 1) {
            dep = ready;
          } else {
            dep = Math.max(dep, ready);
          }
          updated.put(
              trip.id(),
              new TimetableTrip(
                  trip.id(),
                  trip.timetableId(),
                  trip.routeId(),
                  trip.sequence(),
                  trip.tripCode(),
                  Math.floorMod(zero + dep, 86400),
                  trip.dutyId()));
        }
        prevArrival = dep + plan.totalRunSeconds();
      }
      int returnAt = duty.returnRouteId().isPresent() ? prevArrival + TURNAROUND : prevArrival;
      int returnRun =
          duty.returnRouteId().flatMap(tt::routePlan).map(TimetableRoutePlan::totalRunSeconds).orElse(0);
      duties.add(
          new VehicleDuty(
              duty.id(),
              duty.timetableId(),
              duty.sequence(),
              duty.dutyCode(),
              duty.startDepotNodeId(),
              duty.endDepotNodeId(),
              duty.createRouteId(),
              duty.returnRouteId(),
              duty.tripIds(),
              duty.plannedStartSecondOfDay(),
              Math.floorMod(zero + returnAt, 86400),
              Math.floorMod(zero + returnAt + returnRun, 86400),
              duty.closeReason()));
    }
    List<TimetableTrip> trips = new ArrayList<>();
    for (TimetableTrip trip : tt.trips()) {
      trips.add(updated.get(trip.id()));
    }
    return new Timetable(
        tt.id(), tt.companyId(), tt.operatorId(), tt.lineId(), tt.code(), tt.name(), tt.status(),
        tt.zoneId(), tt.serviceStartSecondOfDay(), tt.serviceEndSecondOfDay(), tt.routePlans(),
        trips, duties, tt.notes(), tt.createdAt(), tt.updatedAt());
  }

  private static String top(Map<String, Integer> m, int limit) {
    return m.entrySet().stream()
        .sorted(Map.Entry.<String, Integer>comparingByValue().reversed())
        .limit(limit)
        .map(e -> e.getKey().replace("SURC:", "").replace("SWITCHER:Towny:", "SW") + "=" + e.getValue())
        .toList()
        .toString();
  }

  private static String stats(List<Integer> v) {
    if (v.isEmpty()) return "-";
    List<Integer> s = new ArrayList<>(v);
    s.sort(null);
    long sum = 0;
    for (int x : s) sum += x;
    return String.format(
        Locale.ROOT, "n=%d min=%d med=%d mean=%.0f max=%d", s.size(), s.get(0), s.get(s.size() / 2), sum / (double) s.size(), s.get(s.size() - 1));
  }

  private static int rel(int secondOfDay, int zero) {
    int r = secondOfDay - zero;
    return r < -3600 ? r + 86400 : r;
  }

  private static Object get(Object target, String name) throws Exception {
    Method m = target.getClass().getDeclaredMethod(name);
    m.setAccessible(true);
    return m.invoke(target);
  }

  private static int readWeight(Route route) {
    Object raw = route.metadata() == null ? null : route.metadata().get("spawn_weight");
    if (raw instanceof Number number) {
      return Math.max(0, number.intValue());
    }
    if (raw instanceof String text) {
      try {
        return Math.max(0, Integer.parseInt(text.trim()));
      } catch (NumberFormatException ignored) {
        return 1;
      }
    }
    return 1;
  }
